# Best-effort extras. setup_env.sh installs these but does NOT fail the whole setup
# if one is unavailable on the platform — the scripts degrade gracefully when a tool
# is missing (e.g. score_signatures.py falls back to decoupler if pyUCell is absent).
# Tool-level degradation here is fine; the hard MCP gate (knowledgebase — required;
# zotero is strongly preferred but optional) is enforced by the skill itself, not by
# these packages.
pyucell        # robust rank-based per-cell scoring with negative markers. Current
               # releases (0.5.x) have NO numpy<2 pin and install + run fine on the
               # numpy-2 core venv. score_signatures.py only invokes it when a marker
               # set has negative markers (else decoupler aucell covers per-cell), and
               # guards the import, so its absence is non-fatal.
singler        # SingleR in Python (BiocPy), no R — alternative reference cross-check
celldex        # reference datasets for singler
faiss-cpu      # optional fast kNN backend for cellmapper (pynndescent in core covers CPU)
pertpy         # condition DE / Milo (Job B). NOTE: some releases (e.g. 1.1.0) break
               # against scanpy >=1.11 (import error). No core annotation script imports
               # it; only Job-B condition-DE work does — pin a compatible release then.
pydeseq2       # pseudobulk DE backend (Job B)
seaborn
cellxgene-census  # remote TileDB-SOMA over S3; network-dependent (no atlas download).
               # Used by fetch_census_reference.py to fetch a capped, on-demand reference
               # AnnData for reference_map.py when the user has no atlas of their own.
               # Absent? fetch_census_reference.py exits cleanly with an install hint.
